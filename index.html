    <script>

        // --- DATABASE INIZIALE (LOCAL STORAGE) ---

        let db = JSON.parse(localStorage.getItem('danceflow_db')) || {

            courses: [

                { id: 1, name: "Danza Classica Principianti" },

                { id: 2, name: "Hip Hop Avanzato" }

            ],

            students: [

                { id: 1, name: "Sofia Rossi" },

                { id: 2, name: "Marco Bianchi" },

                { id: 3, name: "Giulia Verdi" }

            ],

            enrollments: [

                { studentId: 1, courseId: 1 },

                { studentId: 2, courseId: 1 },

                { studentId: 3, courseId: 2 }

            ],

            attendance: [] // Struttura: { date: 'YYYY-MM-DD', courseId: X, studentId: Y, status: 'Presente/Assente/Giustificato' }

        };


        function saveDB() {

            localStorage.setItem('danceflow_db', JSON.stringify(db));

        }


        let currentUser = null;

        let currentAttendanceState = {}; // Stato temporaneo della schermata appello corrente


        // --- SISTEMA DI AUTENTICAZIONE ---

        function handleLogin(event) {

            event.preventDefault();

            const user = document.getElementById('username').value.trim();

            

            if (user === 'admin') {

                currentUser = { name: 'Segreteria', role: 'admin' };

            } else if (user === 'teacher') {

                currentUser = { name: 'Insegnante M.', role: 'teacher' };

            } else {

                alert('Utente non riconosciuto. Usa "admin" o "teacher".');

                return;

            }

            

            initSession();

        }


        function initSession() {

            document.getElementById('view-login').classList.add('hidden');

            document.getElementById('navbar').classList.remove('hidden');

            document.getElementById('navbar').classList.add('flex');

            document.getElementById('user-badge').innerText = currentUser.name;


            if (currentUser.role === 'admin') {

                document.getElementById('view-admin').classList.remove('hidden');

                refreshAdminDropdowns();

                renderGlobalReport();

            } else if (currentUser.role === 'teacher') {

                document.getElementById('view-teacher').classList.remove('hidden');

                document.getElementById('teacher-date-select').value = new Date().toISOString().split('T')[0];

                refreshTeacherCourses();

                loadTeacherClasslist();

            }

        }


        function logout() {

            currentUser = null;

            document.getElementById('view-admin').classList.add('hidden');

            document.getElementById('view-teacher').classList.add('hidden');

            document.getElementById('navbar').classList.remove('flex');

            document.getElementById('navbar').classList.add('hidden');

            document.getElementById('view-login').classList.remove('hidden');

            document.getElementById('username').value = '';

            document.getElementById('password').value = '';

        }


        // --- FUNZIONALITÀ SEGRETERIA (ADMIN) ---

        function refreshAdminDropdowns() {

            const sSelect = document.getElementById('enroll-student-select');

            const cSelect = document.getElementById('enroll-course-select');

            

            sSelect.innerHTML = db.students.map(s => `<option value="${s.id}">${s.name}</option>`).join('');

            cSelect.innerHTML = db.courses.map(c => `<option value="${c.id}">${c.name}</option>`).join('');

        }


        function addCourse(e) {

            e.preventDefault();

            const name = document.getElementById('new-course-name').value.trim();

            const id = Date.now();

            db.courses.push({ id, name });

            saveDB();

            document.getElementById('new-course-name').value = '';

            refreshAdminDropdowns();

            alert('Corso inserito con successo!');

        }


        function addStudent(e) {

            e.preventDefault();

            const name = document.getElementById('new-student-name').value.trim();

            const id = Date.now();

            db.students.push({ id, name });

            saveDB();

            document.getElementById('new-student-name').value = '';

            refreshAdminDropdowns();

            alert('Allievo iscritto all\'anagrafica!');

        }


        function enrollStudent(e) {

            e.preventDefault();

            const studentId = parseInt(document.getElementById('enroll-student-select').value);

            const courseId = parseInt(document.getElementById('enroll-course-select').value);


            // Evita duplicati di iscrizione

            const exists = db.enrollments.some(en => en.studentId === studentId && en.courseId === courseId);

            if(exists) return alert('L\'allievo è già associato a questo corso.');


            db.enrollments.push({ studentId, courseId });

            saveDB();

            alert('Associazione completata!');

        }


        function renderGlobalReport() {

            const tbody = document.getElementById('global-report-table');

            if (db.attendance.length === 0) {

                tbody.innerHTML = `<tr><td colspan="4" class="p-4 text-center text-xs text-gray-400">Nessun dato di presenza registrato.</td></tr>`;

                return;

            }


            // Ordina per data decrescente

            const sortedAttendance = [...db.attendance].sort((a,b) => b.date.localeCompare(a.date));


            tbody.innerHTML = sortedAttendance.map(att => {

                const course = db.courses.find(c => c.id === att.courseId)?.name || 'N/D';

                const student = db.students.find(s => s.id === att.studentId)?.name || 'N/D';

                let badgeColor = 'bg-gray-100 text-gray-800';

                if(att.status === 'Presente') badgeColor = 'bg-green-100 text-green-800';

                if(att.status === 'Assente') badgeColor = 'bg-red-100 text-red-800';

                if(att.status === 'Giustificato') badgeColor = 'bg-yellow-100 text-yellow-800';


                return `

                    <tr class="hover:bg-gray-50 text-xs">

                        <td class="p-3 font-medium">${att.date}</td>

                        <td class="p-3">${course}</td>

                        <td class="p-3">${student}</td>

                        <td class="p-3"><span class="px-2 py-0.5 rounded-full font-semibold ${badgeColor}">${att.status}</span></td>

                    </tr>

                `;

            }).join('');

        }


        // --- FUNZIONALITÀ INSEGNANTE ---

        function refreshTeacherCourses() {

            const select = document.getElementById('teacher-course-select');

            select.innerHTML = db.courses.map(c => `<option value="${c.id}">${c.name}</option>`).join('');

        }


        function loadTeacherClasslist() {

            const courseId = parseInt(document.getElementById('teacher-course-select').value);

            const date = document.getElementById('teacher-date-select').value;

            const container = document.getElementById('teacher-classlist');


            if(!courseId || !date) return;


            // Trova allievi iscritti a questo corso

            const enrolledIds = db.enrollments.filter(e => e.courseId === courseId).map(e => e.studentId);

            const classlist = db.students.filter(s => enrolledIds.includes(s.id));


            // Resetta lo stato temporaneo della UI

            currentAttendanceState = {};


            if(classlist.length === 0) {

                container.innerHTML = `<p class="text-sm text-center py-6 text-gray-400">Nessun allievo iscritto a questo corso.</p>`;

                document.getElementById('attendance-counter').innerText = `0 allievi`;

                return;

            }


            document.getElementById('attendance-counter').innerText = `${classlist.length} allievi iscritti`;


            // Carica lo stato preesistente nel DB per quel giorno/corso se presente

            container.innerHTML = classlist.map(student => {

                const existing = db.attendance.find(a => a.date === date && a.courseId === courseId && a.studentId === student.id);

                const currentStatus = existing ? existing.status : 'Assente'; // Di default consideriamo assente non marcato

                currentAttendanceState[student.id] = currentStatus;


                return generateStudentRowHTML(student.id, student.name, currentStatus);

            }).join('');

        }


        function generateStudentRowHTML(id, name, status) {

            return `

                <div class="flex flex-col sm:flex-row sm:items-center justify-between p-3 bg-gray-50 border border-gray-100 rounded-xl space-y-2 sm:space-y-0">

                    <span class="font-medium text-sm text-gray-800">${name}</span>

                    <div class="flex space-x-1.5 justify-end">

                        <button onclick="setStudentStatus(${id}, 'Presente')" id="btn-${id}-Presente" class="status-btn text-xs px-2.5 py-1.5 rounded-lg border font-semibold transition ${status === 'Presente' ? 'bg-green-600 text-white border-green-600' : 'bg-white text-gray-600 border-gray-200'}">P</button>

                        <button onclick="setStudentStatus(${id}, 'Assente')" id="btn-${id}-Assente" class="status-btn text-xs px-2.5 py-1.5 rounded-lg border font-semibold transition ${status === 'Assente' ? 'bg-red-600 text-white border-red-600' : 'bg-white text-gray-600 border-gray-200'}">A</button>

                        <button onclick="setStudentStatus(${id}, 'Giustificato')" id="btn-${id}-Giustificato" class="status-btn text-xs px-2.5 py-1.5 rounded-lg border font-semibold transition ${status === 'Giustificato' ? 'bg-yellow-500 text-white border-yellow-500' : 'bg-white text-gray-600 border-gray-200'}">G</button>

                    </div>

                </div>

            `;

        }


        function setStudentStatus(studentId, status) {

            currentAttendanceState[studentId] = status;

            

            // Aggiorna visivamente i bottoni per quel singolo allievo

            const types = ['Presente', 'Assente', 'Giustificato'];

            types.forEach(t => {

                const btn = document.getElementById(`btn-${studentId}-${t}`);

                if (t === status) {

                    let activeClasses = t === 'Presente' ? 'bg-green-600 text-white border-green-600' : t === 'Assente' ? 'bg-red-600 text-white border-red-600' : 'bg-yellow-500 text-white border-yellow-500';

                    btn.className = `status-btn text-xs px-2.5 py-1.5 rounded-lg border font-semibold transition ${activeClasses}`;

                } else {

                    btn.className = `status-btn text-xs px-2.5 py-1.5 rounded-lg border font-semibold transition bg-white text-gray-600 border-gray-200`;

                }

            });

        }


        function saveAttendance() {

            const courseId = parseInt(document.getElementById('teacher-course-select').value);

            const date = document.getElementById('teacher-date-select').value;


            // Rimuovi vecchi record sovrapponibili prima di salvare i nuovi per evitare duplicati

            db.attendance = db.attendance.filter(a => !(a.date === date && a.courseId === courseId));


            // Push del nuovo stato attuale

            for (let studentId in currentAttendanceState) {

                db.attendance.push({

                    date: date,

                    courseId: courseId,

                    studentId: parseInt(studentId),

                    status: currentAttendanceState[studentId]

                });

            }


            saveDB();

            alert('Presenze salvate correttamente nel registro!');

        }


        // --- REGISTRAZIONE SERVICE WORKER PER PWA ---

        if ('serviceWorker' in navigator) {

            window.addEventListener('load', () => {

                navigator.serviceWorker.register('sw.js')

                    .then(reg => console.log('Service Worker Registrato con successo!', reg.scope))

                    .catch(err => console.log('Errore registrazione Service Worker:', err));

            });

        }
